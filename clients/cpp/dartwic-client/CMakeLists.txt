cmake_minimum_required(VERSION 3.20)
project(DARTWICClient VERSION 3.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
find_package(cppzmq CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(Threads REQUIRED)
set(_tempest "${CMAKE_CURRENT_SOURCE_DIR}/vendor/tempest")
add_library(DARTWICClient STATIC src/dartwic_client.cpp "${_tempest}/src/client_connection.cpp")
add_library(DARTWIC::Client ALIAS DARTWICClient)
set_target_properties(DARTWICClient PROPERTIES EXPORT_NAME Client POSITION_INDEPENDENT_CODE ON)
target_compile_features(DARTWICClient PUBLIC cxx_std_20)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package.json" _client_package)
string(JSON _client_version GET "${_client_package}" version)
target_compile_definitions(DARTWICClient PRIVATE DARTWIC_CLIENT_VERSION="${_client_version}")
string(JSON _minimum_engine GET "${_client_package}" dartwic minEngineVersion)
target_compile_definitions(DARTWICClient PRIVATE DARTWIC_MIN_ENGINE_VERSION="${_minimum_engine}")
target_include_directories(DARTWICClient PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE "${_tempest}/include")
target_link_libraries(DARTWICClient PUBLIC nlohmann_json::nlohmann_json PRIVATE cppzmq Threads::Threads)
add_executable(DARTWICClient_Example EXCLUDE_FROM_ALL examples/analysis.cpp)
target_link_libraries(DARTWICClient_Example PRIVATE DARTWIC::Client)
install(TARGETS DARTWICClient EXPORT DARTWICClientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT DARTWICClientTargets NAMESPACE DARTWIC:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DARTWICClient)
configure_package_config_file(cmake/DARTWICClientConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/DARTWICClientConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DARTWICClient)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/DARTWICClientConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/DARTWICClientConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/DARTWICClientConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/DARTWICClient)
